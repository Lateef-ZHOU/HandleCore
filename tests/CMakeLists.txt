if(HANDLECORE_WITH_PCAP)
    add_executable(pcap_probe pcap/probe.cpp pcap/FakePcap.cpp)
    add_executable(pcap_link_probe pcap/link_probe.cpp)
    foreach(probe IN ITEMS pcap_probe pcap_link_probe)
        set_target_properties(${probe} PROPERTIES AUTOMOC ON)
        target_link_libraries(${probe} PRIVATE Lidar::Pcap)
        target_compile_definitions(${probe} PRIVATE QT_NO_KEYWORDS)
        if(MSVC)
            target_compile_options(${probe} PRIVATE /W4 /WX /utf-8)
        else()
            target_compile_options(${probe} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
        add_test(NAME ${probe} COMMAND ${probe})
        set_tests_properties(${probe} PROPERTIES TIMEOUT 30)
        if(WIN32)
            set_tests_properties(${probe} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:HandlePcapDependency>")
        endif()
    endforeach()
    # Each public Pcap header must compile on its own, without transitive application includes.
    foreach(header IN ITEMS PcapProtocol PcapInterfaces PcapHandle)
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${header}_alone.cpp" CONTENT "#include <${header}.h>\n")
        target_sources(pcap_link_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${header}_alone.cpp")
    endforeach()
endif()
